cmake_minimum_required(VERSION 3.22.1)
project(tonekeep_lyrics LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Preview APKs are debug builds too: unoptimized tensor kernels are unusably slow.
set(CMAKE_C_FLAGS_DEBUG "-O3 -g")
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g")
# Static upstream libraries avoid collisions with the existing FFmpeg/Python runtime.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(whisper.cpp)
add_library(tonekeep_lyrics SHARED lyrics_jni.cpp)
target_link_libraries(tonekeep_lyrics PRIVATE whisper log)
target_link_options(tonekeep_lyrics PRIVATE "-Wl,-z,max-page-size=16384")
